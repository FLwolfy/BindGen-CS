cmake_minimum_required(VERSION 3.25)
project(BGCSUpstreamCorpus LANGUAGES C CXX)

if(NOT IS_DIRECTORY "${BGCS_CORPUS_ROOT}" OR NOT BGCS_CORPUS_LIBRARY)
  message(FATAL_ERROR "BGCS_CORPUS_ROOT and BGCS_CORPUS_LIBRARY must identify the verified upstream checkout.")
endif()
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  add_compile_options(/bigobj /Zc:__cplusplus /Zc:preprocessor /utf-8)
endif()
find_package(Threads REQUIRED)
set(corpus "${BGCS_CORPUS_ROOT}")
set(product "${BGCS_CORPUS_LIBRARY}")

if(product STREQUAL "sdl3")
  set(SDL_SHARED ON CACHE BOOL "" FORCE)
  set(SDL_STATIC OFF CACHE BOOL "" FORCE)
  set(SDL_TESTS OFF CACHE BOOL "" FORCE)
  set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
  set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(SDL_INSTALL OFF CACHE BOOL "" FORCE)
  add_subdirectory("${corpus}/SDL" SDL)
  set(native_target SDL3-shared)
elseif(product STREQUAL "miniaudio")
  add_library(corpus_native SHARED "${corpus}/miniaudio/extras/miniaudio_split/miniaudio.c")
  target_include_directories(corpus_native PUBLIC "${corpus}/miniaudio/extras/miniaudio_split")
  target_compile_definitions(corpus_native PRIVATE MA_DLL MA_IMPLEMENTATION)
  target_link_libraries(corpus_native PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
  if(UNIX)
    target_link_libraries(corpus_native PRIVATE m)
  endif()
  set(native_target corpus_native)
elseif(product STREQUAL "cimgui" OR product STREQUAL "cimguizmo")
  set(imgui_wrapper "${corpus}/cimgui")
  set(imgui "${imgui_wrapper}/imgui")
  add_library(corpus_native SHARED
    "${imgui_wrapper}/cimgui.cpp" "${imgui}/imgui.cpp" "${imgui}/imgui_draw.cpp"
    "${imgui}/imgui_demo.cpp" "${imgui}/imgui_widgets.cpp" "${imgui}/imgui_tables.cpp")
  target_include_directories(corpus_native PUBLIC "${imgui_wrapper}" "${imgui}")
  target_compile_definitions(corpus_native PRIVATE IMGUI_DISABLE_OBSOLETE_FUNCTIONS=1)
  if(EXISTS "${imgui_wrapper}/cimconfig.h")
    target_compile_definitions(corpus_native PRIVATE "IMGUI_USER_CONFIG=\"../cimconfig.h\"")
  endif()
  if(product STREQUAL "cimguizmo")
    target_sources(corpus_native PRIVATE "${corpus}/cimguizmo/cimguizmo.cpp" "${corpus}/cimguizmo/ImGuizmo/ImGuizmo.cpp")
    target_include_directories(corpus_native PUBLIC "${corpus}/cimguizmo")
  endif()
  if(WIN32)
    target_link_libraries(corpus_native PRIVATE imm32)
  endif()
  set(native_target corpus_native)
elseif(product STREQUAL "bgfx" OR product STREQUAL "bimg")
  # This is the complete upstream bimg core library, not fabricated implementations
  # of the methods used by the probe. Its upstream script selects these same units.
  add_library(corpus_bx STATIC "${corpus}/bx/src/amalgamated.cpp")
  target_include_directories(corpus_bx PUBLIC "${corpus}/bx/include" PRIVATE "${corpus}/bx/3rdparty")
  target_compile_definitions(corpus_bx PUBLIC BX_CONFIG_DEBUG=0)
  target_link_libraries(corpus_bx PUBLIC Threads::Threads ${CMAKE_DL_LIBS})
  if(UNIX)
    target_link_libraries(corpus_bx PUBLIC m)
  endif()
  if(WIN32)
    target_include_directories(corpus_bx PUBLIC "${corpus}/bx/include/compat/msvc")
    target_link_libraries(corpus_bx PUBLIC psapi)
  endif()
  if(MSVC)
    # Match bx's upstream NoExceptions build policy. Its Windows thread-naming
    # implementation uses SEH, which cannot share C++ exception unwinding here.
    target_compile_options(corpus_bx PRIVATE /EHs-c-)
    target_compile_definitions(corpus_bx PRIVATE _HAS_EXCEPTIONS=0)
  else()
    target_compile_options(corpus_bx PRIVATE -fno-exceptions)
  endif()
  add_library(corpus_bimg STATIC "${corpus}/bimg/src/image.cpp" "${corpus}/bimg/src/image_gnf.cpp" "${corpus}/bimg/src/image_decode.cpp")
  file(GLOB astc_sources CONFIGURE_DEPENDS "${corpus}/bimg/3rdparty/astc-encoder/source/*.cpp")
  target_sources(corpus_bimg PRIVATE ${astc_sources})
  target_include_directories(corpus_bimg PUBLIC "${corpus}/bimg/include" PRIVATE "${corpus}/bimg/3rdparty/astc-encoder/include"
    "${corpus}/bimg/3rdparty" "${corpus}/bimg/3rdparty/tinyexr/deps")
  target_link_libraries(corpus_bimg PUBLIC corpus_bx)
  if(product STREQUAL "bgfx")
    # The upstream Noop renderer exercises CPU-side real bgfx APIs without opening
    # a window or requiring an interactive GPU on a build runner.
    add_library(corpus_native SHARED "${corpus}/bgfx/src/amalgamated.cpp")
    target_include_directories(corpus_native PUBLIC "${corpus}/bgfx/include" PRIVATE "${corpus}/bgfx/3rdparty")
    target_compile_definitions(corpus_native PRIVATE BGFX_SHARED_LIB_BUILD=1 BGFX_CONFIG_MULTITHREADED=0
      BGFX_CONFIG_RENDERER_AGC=0 BGFX_CONFIG_RENDERER_DIRECT3D11=0 BGFX_CONFIG_RENDERER_DIRECT3D12=0
      BGFX_CONFIG_RENDERER_GNM=0 BGFX_CONFIG_RENDERER_METAL=0 BGFX_CONFIG_RENDERER_NVN=0
      BGFX_CONFIG_RENDERER_OPENGL=0 BGFX_CONFIG_RENDERER_OPENGLES=0 BGFX_CONFIG_RENDERER_VULKAN=0 BGFX_CONFIG_RENDERER_WEBGPU=0)
    target_link_libraries(corpus_native PUBLIC corpus_bimg)
    if(WIN32)
      target_link_libraries(corpus_native PRIVATE gdi32)
    endif()
    set(native_target corpus_native)
  elseif(BGCS_BRIDGE_SOURCES)
    add_library(corpus_native SHARED ${BGCS_BRIDGE_SOURCES})
    target_include_directories(corpus_native PUBLIC ${BGCS_BRIDGE_INCLUDE_DIRECTORIES})
    target_link_libraries(corpus_native PUBLIC corpus_bimg)
    set(native_target corpus_native)
  else()
    set(native_target corpus_bimg)
  endif()
else()
  message(FATAL_ERROR "The requested corpus product is not registered: ${product}")
endif()

if(BGCS_ADDITIONAL_SOURCES)
  # Compile the public contract and helpers with the same compiler/SDK and public
  # usage requirements, without upstream implementation-only defines such as
  # SDL_INTERNAL or MA_IMPLEMENTATION changing the public header definitions.
  # Their actual objects still become part of the one configured native module.
  add_library(corpus_adapters OBJECT ${BGCS_ADDITIONAL_SOURCES})
  target_include_directories(corpus_adapters PRIVATE ${BGCS_ADDITIONAL_INCLUDE_DIRECTORIES}
    $<TARGET_PROPERTY:${native_target},INTERFACE_INCLUDE_DIRECTORIES>)
  target_compile_definitions(corpus_adapters PRIVATE
    $<TARGET_PROPERTY:${native_target},INTERFACE_COMPILE_DEFINITIONS>)
  target_sources(${native_target} PRIVATE $<TARGET_OBJECTS:corpus_adapters>)
endif()

if(TARGET corpus_native)
  set_target_properties(corpus_native PROPERTIES OUTPUT_NAME "${product}" PREFIX "")
endif()
add_executable(corpus_probe "${CMAKE_CURRENT_SOURCE_DIR}/${product}.cpp")
target_link_libraries(corpus_probe PRIVATE ${native_target})
if(product STREQUAL "bimg" AND BGCS_BRIDGE_SOURCES)
  target_link_libraries(corpus_probe PRIVATE corpus_bimg)
endif()
if(WIN32)
  add_custom_command(TARGET corpus_probe POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_if_different
    "$<TARGET_FILE:${native_target}>" "$<TARGET_FILE_DIR:corpus_probe>" VERBATIM)
endif()
set(upstream_static_assets "")
if(product STREQUAL "bimg")
  set(upstream_static_assets ",\"upstreamStaticLibraries\":[\"$<TARGET_FILE:corpus_bimg>\",\"$<TARGET_FILE:corpus_bx>\"]")
endif()
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/native-assets-$<CONFIG>.json" CONTENT
  "{\"product\":\"${product}\",\"nativeBinary\":\"$<TARGET_FILE:${native_target}>\",\"probe\":\"$<TARGET_FILE:corpus_probe>\",\"cCompiler\":\"${CMAKE_C_COMPILER}\",\"cppCompiler\":\"${CMAKE_CXX_COMPILER}\",\"cppCompilerVersion\":\"${CMAKE_CXX_COMPILER_VERSION}\",\"system\":\"${CMAKE_SYSTEM_NAME}\",\"processor\":\"${CMAKE_SYSTEM_PROCESSOR}\"${upstream_static_assets}}\n")
